func local_contains(p):
	var b=config.bounds;return p.x>b[0] and p.x<b[2] and p.y>b[1] and p.y<b[3]

func coastal_sample(lon,lat):
	var p=world.project(lon,lat);var point=Vector2(p.x,p.z)
	if local_contains(point):return super.coastal_sample(lon,lat)
	return catalog.nearest(point) if catalog!=null else null
func plant_clearance(sample):return lerp(.006,.035,float(sample.get("beach",1.)))
func profile_sample_height(sample):
	if not sample.has("beach"):return super.profile_sample_height(sample)
	return styled_height(sample.point,sample.distance,sample.beach)
func styled_height(point,d,beach_factor):
	var cut=lerp(.008,.10,beach_factor)
	if d<=0.:return d*lerp(.7,.08,beach_factor)
	var geo=world.unproject(Vector3(point.x,0,point.y));var original=world.elevation(geo.x,geo.y)+.00015
	# Nonnegative land profile: rocky faces rise to the existing terrain within
	# eight metres, while a low-gradient beach has a longer graded backshore.
	return lerp(d*.055,original,smoothstep(cut*.25,cut,d))
func clip_edge(a,b,minimum,maximum):
	var lo=0.;var hi=1.;var delta=b-a
	for axis in 2:
		if abs(delta[axis])<.000001:
			if a[axis]<minimum[axis] or a[axis]>maximum[axis]:return null
		else:
			var u=(minimum[axis]-a[axis])/delta[axis];var v=(maximum[axis]-a[axis])/delta[axis]
			lo=max(lo,min(u,v));hi=min(hi,max(u,v))
	if lo>=hi:return null
	return Vector2(lo,hi)
func request_patch(center,span,key):
	busy=true;requested_key=key;generation+=1;var job=generation
	var minimum=center-Vector2.ONE*span*.5;var maximum=center+Vector2.ONE*span*.5
	var found={}
	for y in range(floori(minimum.y-.6),ceili(maximum.y+.6)+1):
		for x in range(floori(minimum.x-.6),ceili(maximum.x+.6)+1):
			for id in catalog.ids_at(Vector2(x,y)):found[id]=true
	# In exceptionally dense archipelagos reduce the local footprint rather
	# than dropping islands or cutting holes where no coast mesh is drawn.
	if found.size()>14000 and span>4.:
		busy=false;request_patch(center,span*.5,key);return
	var edges=[];var total=0.;var clock=Time.get_ticks_usec()
	var sorted=found.keys();sorted.sort()
	for id in sorted:
		var s=catalog.segment(id);var a=Vector2(s[0],s[1]);var b=Vector2(s[2],s[3]);var clipped=clip_edge(a,b,minimum-Vector2.ONE*.25,maximum+Vector2.ONE*.25)
		if clipped==null:continue
		var start=a.lerp(b,clipped.x);var end=a.lerp(b,clipped.y);var len=start.distance_to(end)
		if len<.001:continue
		var normal=Vector2(s[4],s[5]);var middle=(start+end)*.5;var geo=world.unproject(Vector3(middle.x,0,middle.y));var h=world.elevation(geo.x,geo.y)
		var beach_factor=min(float(s[8]),1.-smoothstep(.008,.025,h));var rock=max(float(s[9]),smoothstep(.008,.025,h))
		var opposite=catalog.nearest(middle-normal*.22);var exposure=.15 if opposite!=null and opposite.distance>.001 else 1.
		edges.append({"a":start,"b":end,"normal":normal,"along":s[6]+s[7]*clipped.x,"len":len,"beach":beach_factor,"rock":rock,"exposure":exposure});total+=len
		if Time.get_ticks_usec()-clock>3500:
			await get_tree().process_frame;clock=Time.get_ticks_usec()
			if job!=generation:busy=false;return
	if edges.is_empty():
		patch_key=key;global_meshes.clear();beach.mesh=null;surf.mesh=null;built=false;set_active(false);busy=false;return
	var spacing=max(.012,total/2500.);var rows=0
	# Shared miter joins prevent open wedges between adjacent coast segments.
	for i in edges.size():
		var edge=edges[i];var before=edges[(i-1+edges.size())%edges.size()];var after=edges[(i+1)%edges.size()]
		edge["start_normal"]=edge.normal;edge["end_normal"]=edge.normal
		edge["start_style"]=Vector2(edge.beach,edge.rock);edge["end_style"]=edge.start_style
		for side in ["start","end"]:
			var adjacent=before if side=="start" else after
			var connected=edge.a.distance_to(adjacent.b)<.001 if side=="start" else edge.b.distance_to(adjacent.a)<.001
			if connected and edge.normal.dot(adjacent.normal)>.0:
				var normal=(edge.normal+adjacent.normal).normalized();edge[side+"_normal"]=normal/max(.5,normal.dot(edge.normal))
				edge[side+"_style"]=(Vector2(edge.beach,edge.rock)+Vector2(adjacent.beach,adjacent.rock))*.5
	for edge in edges:rows+=maxi(1,ceili(edge.len/spacing))+1
	var columns=mini(45,maxi(5,160000/maxi(1,rows)))
	if rows*columns>180000:
		busy=false;request_patch(center,span*.5,key);return
	var vertices=PackedVector3Array();var water_vertices=PackedVector3Array();var normals=PackedVector3Array();var uv=PackedVector2Array();var colors=PackedColorArray();var mesh_indices=PackedInt32Array()
	var data=PackedFloat32Array();var bins=[]
	for i in 1024:bins.append(PackedInt32Array())
	var style_counts=[0,0,0]
	for edge in edges:
		var first=vertices.size();var steps=maxi(1,ceili(edge.len/spacing));var cut=lerp(.008,.10,edge.beach);var n=edge.normal;var id=data.size()/12
		var a=edge.a-center;var b=edge.b-center
		data.append_array(PackedFloat32Array([a.x,a.y,b.x,b.y,n.x,n.y,fmod(edge.along,4.096),edge.len*(-1. if edge.exposure<.5 else 1.),edge.start_style.x,edge.start_style.y,edge.end_style.x,edge.end_style.y]))
		var low=((Vector2(min(a.x,b.x),min(a.y,b.y))-Vector2.ONE*.55+Vector2.ONE*span*.5)/span*32.).floor().clamp(Vector2.ZERO,Vector2.ONE*31)
		var high=((Vector2(max(a.x,b.x),max(a.y,b.y))+Vector2.ONE*.55+Vector2.ONE*span*.5)/span*32.).floor().clamp(Vector2.ZERO,Vector2.ONE*31)
		for y in range(int(low.y),int(high.y)+1):
			for x in range(int(low.x),int(high.x)+1):bins[y*32+x].append(id)
		style_counts[0 if edge.beach>.5 else (1 if edge.rock>.5 else 2)]+=1
		for row in steps+1:
			var t=float(row)/steps;var point=edge.a.lerp(edge.b,t);var offset=edge.start_normal.lerp(edge.end_normal,t);var style=edge.start_style.lerp(edge.end_style,t);cut=lerp(.008,.10,style.x)
			# Source terrain varies slowly across this narrow strip. Sample its
			# joining edge twice per row instead of querying every beach vertex.
			var join=point+offset*cut;var end=point+offset*(cut+.020)
			var join_geo=world.unproject(Vector3(join.x,0,join.y));var end_geo=world.unproject(Vector3(end.x,0,end.y))
			var join_height=world.elevation(join_geo.x,join_geo.y)+.0005;var end_height=world.elevation(end_geo.x,end_geo.y)+.0005
			for col in columns:
				var u=float(col)/(columns-1);var d=lerp(-.18,0.,u/.4) if u<.4 else lerp(0.,cut+.020,(u-.4)/.6)
				var p=point+offset*d;var original=max(.001,lerp(join_height,end_height,(d-cut)/.020))
				var h=d*lerp(.7,.08,style.x) if d<=0. else lerp(d*.055,original,smoothstep(cut*.25,cut,d))
				vertices.append(Vector3(p.x,h,p.y));water_vertices.append(Vector3(p.x,max(0.,h),p.y));normals.append(Vector3.UP);uv.append(Vector2(edge.along+t*edge.len,d));colors.append(Color(style.x,style.y,edge.exposure,1.))
			if Time.get_ticks_usec()-clock>3500:
				await get_tree().process_frame;clock=Time.get_ticks_usec()
				if job!=generation:busy=false;return
		for row in steps:
			for col in columns-1:
				var index=first+row*columns+col;mesh_indices.append_array(PackedInt32Array([index,index+1,index+columns,index+1,index+columns+1,index+columns]))
		for row in steps+1:
			for col in columns:
				var left=maxi(0,col-1);var right=mini(columns-1,col+1);var l=vertices[first+row*columns+left];var r=vertices[first+row*columns+right];var distance=Vector2(r.x-l.x,r.z-l.z).length();var slope=(r.y-l.y)/max(.00001,distance)
				normals[first+row*columns+col]=Vector3(-n.x*slope,1.,-n.y*slope).normalized()
	var arrays=[];arrays.resize(Mesh.ARRAY_MAX);arrays[Mesh.ARRAY_VERTEX]=vertices;arrays[Mesh.ARRAY_NORMAL]=normals;arrays[Mesh.ARRAY_TEX_UV]=uv;arrays[Mesh.ARRAY_COLOR]=colors;arrays[Mesh.ARRAY_INDEX]=mesh_indices
	var mesh=ArrayMesh.new();mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES,arrays);arrays[Mesh.ARRAY_VERTEX]=water_vertices;var water_mesh=ArrayMesh.new();water_mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES,arrays)
	var packed=PackedInt32Array();packed.resize(2048);var lookup=PackedInt32Array()
	for i in 1024:packed[i*2]=lookup.size();packed[i*2+1]=bins[i].size();lookup.append_array(bins[i])
	packed.append_array(lookup)
	var bounds=Vector4(minimum.x,minimum.y,span,span)
	var stats={"segments":edges.size(),"vertices":vertices.size(),"beachSegments":style_counts[0],"rockSegments":style_counts[1],"bankSegments":style_counts[2]}
	RenderingServer.call_on_render_thread(field.bake.bind(data,packed,Vector4(-span*.5,-span*.5,span,span),Callable(self,"field_ready").bind(job,[mesh,water_mesh],bounds,key,stats)))
func field_ready(texture,job,meshes,bounds,key,stats):
	busy=false
	if texture==null or job!=generation:return
	global_mode=true;field_texture=texture;patch_bounds=bounds;patch_key=key;global_meshes=meshes;beach.mesh=meshes[0];surf.mesh=meshes[1];built=true
	for entry in materials:
		var mat=entry.get_ref()
		if mat!=null:bind_global(mat)
	metrics.patches+=1;metrics.maxVertices=max(metrics.maxVertices,stats.vertices);metrics.merge(stats,true);metrics.released=false
	set_active(true);print("WORLD_COAST_READY ",key," ",JSON.stringify(stats))
func _process(delta):
	var p=world.project(world.focus.x,world.focus.y);var point=Vector2(p.x,p.z)
	if local_contains(point):
		if busy:generation+=1;requested_key=""
		if global_mode or (not built and not local_meshes.is_empty()):activate_local()
		super._process(delta)
		if built and local_meshes.is_empty():local_meshes.assign([beach.mesh,surf.mesh])
		return
	if world.distance>=12.:
		if active or not global_meshes.is_empty():set_active(false);global_meshes.clear();beach.mesh=null;surf.mesh=null;built=false;patch_key="";metrics.released=true
		if busy:generation+=1
		return
	if not global_mode and active:set_active(false)
	var span=clamp(world.distance*3.,8.,32.);var snap=span*.20;var center=point.snapped(Vector2.ONE*snap);var key=str(center)+"/"+str(snapped(span,.1))
	if key!=patch_key and not busy and world.queue.is_empty():request_patch(center,span,key)
	if global_mode:
		var inside=point.x>patch_bounds.x and point.y>patch_bounds.y and point.x<patch_bounds.x+patch_bounds.z and point.y<patch_bounds.y+patch_bounds.w
		if active!=(built and inside):set_active(built and inside)
	if active:
		beach.material_override.set_shader_parameter("water_time",world.water_time);surf.material_override.set_shader_parameter("water_time",world.water_time)
func is_ready():return not busy
func _exit_tree():
	generation+=1
	RenderingServer.call_on_render_thread(field.release)
