vec3 political_land_color(vec2 geo,vec2 world_xz){
 vec3 color=political_neutral.rgb;
 // Externally supplied scenario cartography. Territory identity and control are
 // separate: a palette update recolors/transfers a region without editing land.
 // Transparent coverage and missing layers always resolve to neutral.
 vec2 q=(geo-political_layer_bounds.xy)/political_layer_bounds.zw;
 q.y=1.-q.y;
 bool eastern=reference_balkans_scope(geo)>.001;
 bool local=geo.x>=6.&&geo.x<=14.&&geo.y>=43.&&geo.y<=47.;
 bool sicily=!local&&reference_rollout_scope(geo)>.001;
 if(sicily)q=vec2((geo.x-6.)/14.,(49.-geo.y)/14.);
 if(local)q=vec2((geo.x-6.)/8.,(47.-geo.y)/4.);
 if(eastern)q=vec2((geo.x-14.)/11.,(46.-geo.y)/10.);
 if(political_layers_enabled&&all(greaterThanEqual(q,vec2(0.)))&&all(lessThanEqual(q,vec2(1.)))){
  vec4 layer;
  vec3 encoded=sicily?reference_detail(q,eastern).rgb:(local?texture(reference_local_ids,q).rgb:texture(political_boundary_layer,q).rgb);
  if(political_ids_mode){layer=reference_owner(q,local,sicily,eastern);}else{layer=texture(political_color_layer,q);}
  color=mix(color,layer.rgb,layer.a);
  if(political_boundaries_enabled){
   float boundary=encoded.r;float strength=.6;
   if(political_ids_mode){
    if(reference_rollout_scope(geo)>.5){boundary=eastern?0.:reference_antialiased_edge(q,local,sicily,eastern);strength=eastern?0.:.62;}else{
    // Continuous 2.4-pixel seams in both directions at every zoom/rotation.
    // Confidence remains in the roster/data, rather than hiding tribal edges.
    vec2 dx=dFdx(q)*1.2,dy=dFdy(q)*1.2;
    boundary=max(max(political_owner_edge(layer,reference_owner(q+dx,local,sicily,eastern)),political_owner_edge(layer,reference_owner(q-dx,local,sicily,eastern))),max(political_owner_edge(layer,reference_owner(q+dy,local,sicily,eastern)),political_owner_edge(layer,reference_owner(q-dy,local,sicily,eastern))));
    boundary*=step(.01,layer.a);
    // A second, wider engraved rim separates deep faction fields from the seam.
    vec2 hx=dx*1.9,hy=dy*1.9;
    float rim=max(max(political_owner_edge(layer,reference_owner(q+hx,local,sicily,eastern)),political_owner_edge(layer,reference_owner(q-hx,local,sicily,eastern))),max(political_owner_edge(layer,reference_owner(q+hy,local,sicily,eastern)),political_owner_edge(layer,reference_owner(q-hy,local,sicily,eastern))));
    color=mix(color,mix(color,vec3(.57,.49,.34),.18),rim*(1.-boundary)*step(.01,layer.a));
    strength=.80;
    }
   }
   color=mix(color,vec3(.028,.026,.032),boundary*strength);
  }
 }
 // Restrained continuous engraving grain, with no invented geographic divisions.
 float grain=sin(world_xz.x*.27+sin(world_xz.y*.13))*sin(world_xz.y*.21+sin(world_xz.x*.17));
 return color*(.985+grain*.015);
}

	float weight=alpine_scope(geo);if(weight<=0.)return 0.;
